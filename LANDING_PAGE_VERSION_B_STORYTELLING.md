## 4. UI/UX 특수 요소

### 4.1 타이핑 애니메이션

```javascript
// 파일 열기 시뮬레이션
const files = [
    'header.php 열기...',
    'footer.php 열기...',
    'style.css 열기...',
    'page-template-1.php 열기...',
    // ...
];

function typeFiles(element) {
    let fileIndex = 0;
    let charIndex = 0;

    function type() {
        if (fileIndex < files.length) {
            if (charIndex < files[fileIndex].length) {
                element.innerHTML += files[fileIndex].charAt(charIndex);
                charIndex++;
                setTimeout(type, 30);
            } else {
                element.innerHTML += '<br>';
                fileIndex++;
                charIndex = 0;
                setTimeout(type, 200);
            }
        }
    }

    type();
}
```

### 4.2 스크롤 기반 스토리 진행

```javascript
// 스크롤에 따른 스토리 요소 등장
const storyElements = document.querySelectorAll('.story-element');

const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
        if (entry.isIntersecting) {
            entry.target.classList.add('reveal');

            // 순차적 등장 효과
            const delay = entry.target.dataset.delay || 0;
            entry.target.style.transitionDelay = `${delay}ms`;
        }
    });
}, {
    threshold: 0.3,
    rootMargin: '-100px 0px'
});

storyElements.forEach(el => observer.observe(el));
```

### 4.4 카운터 증가 효과

```javascript
// 파일 개수 카운터
function countUp(element, target, duration = 1000) {
    let start = 0;
    const step = target / (duration / 16);

    function update() {
        start += step;
        if (start < target) {
            element.textContent = Math.floor(start) + '개 파일';
            requestAnimationFrame(update);
        } else {
            element.textContent = target + '개 파일 수정 필요';
            element.classList.add('complete');
        }
    }

    update();
}
```
